%% 🔹 1. Select audio file (NO PATH ISSUES)
[file, path] = uigetfile({'*.wav;*.mp3'}, 'Select an audio file');

if isequal(file,0)
    disp("❌ No file selected");
    return;
end

[x, Fs] = audioread(fullfile(path, file));

%% 🔹 2. Convert to mono (if stereo)
if size(x,2) > 1
    x = mean(x,2);
end

x = x';   % row vector

%% 🔹 3. Normalize signal
x = x - mean(x);
x = x / (max(abs(x)) + 1e-9);

t = (0:length(x)-1)/Fs;

%% 🔹 4. Choose carrier frequency
fc = 400;   % Hz (audible carrier)

%% 🔹 5. Modulation index
mu = 0.7;   % keep between 0 and 1

%% 🔹 6. AM modulation
x_am = (1 + mu*x) .* cos(2*pi*fc*t);

% Normalize again
x_am = x_am / (max(abs(x_am)) + 1e-9);

%% 🔹 7. Play signals
disp("▶ Playing ORIGINAL audio...");
sound(x, Fs);
pause(length(x)/Fs + 1);

disp("▶ Playing AM MODULATED audio...");
sound(x_am, Fs);

%% 🔹 8. Save output file
output_file = fullfile(path, 'am_audio.wav');
audiowrite(output_file, x_am, Fs);

disp("✅ AM audio saved as:");
disp(output_file);

%% 🔹 9. Plot (optional)
figure;
subplot(2,1,1);
plot(t, x);
title("Original Audio");
xlabel("Time (s)"); ylabel("Amplitude"); grid on;

subplot(2,1,2);
plot(t, x_am);
title("AM Modulated Audio");
xlabel("Time (s)"); ylabel("Amplitude"); grid on;
